\section{Аннотация}
В работе исследуются свойства постоянных неодимовых магнитов шарообразной
формы и с их помощью измеряются горизонтальная и вертикальная составляющие
индукции магнитного поля Земли и магнитное наклонение.

Магнитный момент шарика определяется двумя методами: по максимальному
расстоянию, на котором два шарика удерживают друг друга в поле тяжести, и
по силе сцепления шариков (по весу отрывающейся цепочки). По нему
рассчитываются намагниченность и остаточная индукция материала NdFeB.
Горизонтальная составляющая поля Земли находится по периоду крутильных
колебаний «магнитной стрелки» из $n$ шариков, вертикальная --- по
моменту силы тяжести груза, уравновешивающего стрелку в горизонтальном
положении. Линейность зависимостей $T(n)$ и $\mathcal{M}(n)$ проверяет
аддитивность магнитных моментов шариков. Ожидается, что
остаточная индукция окажется порядка $1$~Тл, а параметры поля Земли
согласуются по порядку величины со справочными для Московского региона.

\section{Теоретические сведения}
Расчёт ведётся в гауссовой системе (СГС), как в методическом пособии;
итоговые значения индукции переводятся в СИ ($1~\text{Гс} = 10^{-4}~\text{Тл}
= 100$~мкТл, $1~\text{эрг/Гс} = 10^{-3}~\text{А}\cdot\text{м}^2$).

\subsection{Точечный магнитный диполь}
Поле точечного диполя с моментом $\boldsymbol{\mm}$
\begin{equation}
    \label{eq:dipole}
    \mathbf{B} = \frac{3(\boldsymbol{\mm}\cdot\mathbf{r})\mathbf{r}}{r^5}
                 - \frac{\boldsymbol{\mm}}{r^3}
    \qquad \left(\text{в СИ домножить на } \tfrac{\mu_0}{4\pi}\right).
\end{equation}
Во внешнем поле на диполь действуют момент сил и (в неоднородном поле) сила
\begin{equation}
    \label{eq:torque}
    \boldsymbol{\mathcal{M}} = [\boldsymbol{\mm}\times\mathbf{B}],
    \qquad
    W = -(\boldsymbol{\mm}\cdot\mathbf{B}),
    \qquad
    \mathbf{F} = -\nabla W = (\boldsymbol{\mm}\cdot\nabla)\mathbf{B}.
\end{equation}
Два диполя, направленных вдоль соединяющей их прямой, притягиваются (при
сонаправленных моментах) с силой
\begin{equation}
    \label{eq:F12}
    F_{12} = \frac{6\mm_1\mm_2}{r^4}.
\end{equation}

\subsection{Однородно намагниченный шар}
Вне однородно намагниченного шара радиуса $R$ поле совпадает с полем
точечного диполя в его центре, внутри шара поле однородно:
\begin{equation}
    \label{eq:B0ball}
    B_0 = \frac{2\mm}{R^3}.
\end{equation}
Намагниченность $M$ и остаточная индукция $B_r$ материала:
\begin{equation}
    \label{eq:magnetization}
    \mm = MV, \quad V = \frac{4}{3}\pi R^3, \qquad B_r = 4\pi M,
\end{equation}
индукция на полюсах шара
\begin{equation}
    \label{eq:Bp}
    B_p = B_0 = \frac{2}{3}B_r.
\end{equation}
Существенно, что материал магнитожёсткий: моменты шариков не меняются во
внешних полях и при контакте друг с другом, поэтому магнитный момент
системы шариков равен сумме их моментов.

\subsection{Определение магнитного момента шарика}
\paragraph{Метод А.} На предельном расстоянии $r_\text{max}$ между центрами
шариков сила притяжения \cref{eq:F12} равна весу нижнего шарика массы $m$:
\begin{equation}
    \label{eq:methodA}
    \mm = \sqrt{\frac{mg\,r_\text{max}^4}{6}}.
\end{equation}

\paragraph{Метод Б.} Сила сцепления двух соприкасающихся шариков ($r = 2R$)
\begin{equation}
    \label{eq:F0}
    F_0 = \frac{6\mm^2}{(2R)^4} = \frac{3\mm^2}{8R^4}.
\end{equation}
Верхний шарик удерживает цепочку, взаимодействуя с несколькими ближайшими
шариками ($F\propto 1/r^4$), поэтому вес отрывающейся цепочки
\begin{equation}
    \label{eq:chain}
    F = F_0\left(1 + \frac{1}{2^4} + \frac{1}{3^4} + \frac{1}{4^4} + \ldots\right)
      \approx 1{,}08\,F_0,
    \qquad
    \mm = \sqrt{\frac{8R^4 F_0}{3}}.
\end{equation}

\subsection{Горизонтальная составляющая поля Земли}
«Стрелка» из $n$ сцепленных шариков, подвешенная горизонтально на
$\Lambda$-образном подвесе, совершает крутильные колебания под действием
момента $\mathcal{M} = -n\mm B_\parallel\sin\theta$. При малых углах и
пренебрежимой упругости нити
\begin{equation}
    \label{eq:Tn_general}
    T = 2\pi\sqrt{\frac{J_n}{n\mm B_\parallel}},
    \qquad
    J_n \approx \frac{1}{12}(nm)(2nR)^2 = \frac{1}{3}n^3 mR^2,
\end{equation}
откуда период пропорционален числу шариков:
\begin{equation}
    \label{eq:Tn}
    T_n = k_T\,n, \qquad k_T = 2\pi\sqrt{\frac{mR^2}{3\mm B_\parallel}}
    \quad\Rightarrow\quad
    B_\parallel = \frac{4\pi^2 mR^2}{3\mm k_T^2}.
\end{equation}

\subsection{Вертикальная составляющая и наклонение}
Стрелка из чётного числа шариков, подвешенная за середину в одной точке,
наклоняется под действием вертикальной составляющей $B_\perp$. Её
выравнивают грузом массы $m_\text{гр}$ на плече $r_\text{гр}$:
\begin{equation}
    \label{eq:moment}
    \mathcal{M}_n = m_\text{гр}\,g\,r_\text{гр} = n\mm B_\perp
    \quad\Rightarrow\quad
    B_\perp = \frac{k_\mathcal{M}}{\mm},
\end{equation}
где $k_\mathcal{M}$ --- наклон прямой $\mathcal{M}(n)$. Магнитное
наклонение и полная индукция
\begin{equation}
    \label{eq:beta}
    \operatorname{tg}\beta = \frac{B_\perp}{B_\parallel},
    \qquad
    B = \sqrt{B_\parallel^2 + B_\perp^2}.
\end{equation}
Если считать Землю однородно намагниченным вдоль оси вращения шаром
(диполь в центре), то на широте $\varphi$
\begin{equation}
    \label{eq:earth_dipole}
    B_\parallel = \frac{\mu_0}{4\pi}\frac{\mm_\text{З}\cos\varphi}{R_\text{З}^3},
    \qquad
    B_\perp = \frac{\mu_0}{4\pi}\frac{2\mm_\text{З}\sin\varphi}{R_\text{З}^3},
    \qquad
    \operatorname{tg}\beta = 2\operatorname{tg}\varphi.
\end{equation}

\subsection{Основные допущения}
\begin{itemize}
    \item шарики однородно намагничены и магнитожёстки (моменты аддитивны);
    \item в методе Б учитываются 4 члена ряда \cref{eq:chain};
    \item момент инерции стрелки --- как у тонкого стержня
          (\cref{eq:Tn_general}), упругость нити не учитывается;
    \item колебания малые ($\sin\theta\approx\theta$);
    \item поле Земли в месте опыта однородно и не искажено посторонними
          магнитными предметами.
\end{itemize}

\begin{enumerate}
    \item Штангенциркулем измеряется диаметр $d = 2R$ шариков, на весах
          (через немагнитную подставку) --- масса $m$; магнитометром
          измеряется индукция на полюсе $B_p$.
    \item \textbf{Метод А} (\cref{fig:setup}а): между шариками
          прокладывается немагнитный брусок и подкладываются листы бумаги до
          тех пор, пока нижний шарик ещё удерживается; измеряется зазор $h$,
          $r_\text{max} = h + d$.
    \item \textbf{Метод Б}: к цепочке из 20--30 шариков подвешивается гиря
          (общая масса $\sim500$~г); добавляя и убирая шарики, находят
          минимальную массу $M$ отрывающейся системы, $F = Mg$.
    \item \textbf{Горизонтальная составляющая} (\cref{fig:setup}б):
          для $n = 3,\dots,12$ после юстировки стрелки в горизонтальное
          положение измеряется время $t$ нескольких колебаний,
          $T = t/N_\text{кол}$. Влияние упругости нити оценивается по
          колебаниям стрелки, свёрнутой в кольцо ($\mm_n = 0$).
    \item \textbf{Вертикальная составляющая} (\cref{fig:setup}в):
          для $n = 4,6,\dots,12$ стрелка, подвешенная за середину,
          уравновешивается кусочками проволоки; на весах определяется масса
          груза $m_\text{гр}$, линейкой --- плечо $r_\text{гр}$.
\end{enumerate}

Обработка данных выполняется скриптом
\filename{scripts/process_data.py}.
Коэффициенты прямых $T(n)$ и $\mathcal{M}(n)$ находятся методом наименьших
квадратов (через начало координат, как предсказывает теория; для контроля
--- с учётом свободного члена). Погрешности косвенных величин:
\begin{gather}
    \label{eq:errors}
    \varepsilon_{\mm}^{(\text{А})} = \sqrt{\left(\tfrac12\varepsilon_m\right)^2 + (2\varepsilon_{r})^2},
    \qquad
    \varepsilon_{\mm}^{(\text{Б})} = \sqrt{\left(\tfrac12\varepsilon_M\right)^2 + (2\varepsilon_{R})^2},
    \\
    \varepsilon_{B_\parallel} = \sqrt{\varepsilon_m^2 + 4\varepsilon_R^2 + \varepsilon_\mm^2 + 4\varepsilon_{k_T}^2},
    \qquad
    \varepsilon_{B_\perp} = \sqrt{\varepsilon_{k_\mathcal{M}}^2 + \varepsilon_\mm^2}. \notag
\end{gather}
Внимание: магнитные предметы (ножницы, металлическая линейка, телефоны)
вблизи установки искажают результат.

    M = \res{Mag}~\text{Гс},
    \qquad
    B_r = \res{Br}~\text{Тл},
    \qquad
    B_p^\text{расч} = \res{Bpcalc}~\text{мТл}.
\end{equation}
Табличное значение для NdFeB: $B_r \approx 1{,}0\div1{,}4$~Тл.
Расчётное значение $B_p$ отличается от измеренного на $\res{Bpdev}$\,\%.

\section{Обсуждение результатов}
\begin{itemize}
    \item Сравнение методов А и Б определения $\mm$: \fillme{}
          (какой точнее и почему; в методе А $r_\text{max}$ входит в
          четвёртой степени).
    \item Остаточная индукция $B_r = \res{Br}$~Тл по сравнению с табличной
          для NdFeB; согласие расчётного и измеренного $B_p$: \fillme.
    \item Линейность $T(n)$ и $\mathcal{M}(n)$ и свободные члены прямых ---
          проверка аддитивности магнитных моментов и магнитожёсткости
          материала: \fillme.
    \item Влияние упругости нити (опыт с кольцом) и приближения тонкого
          стержня для $J_n$ при малых $n$: \fillme.
    \item Сравнение $B_\parallel$, $B_\perp$, $\beta$ со справочными значениями;
          влияние искажения поля в здании и посторонних магнитных
          предметов: \fillme.
    \item Применимость дипольной модели Земли (несовпадение магнитной и
          географической осей): \fillme.
\end{itemize}

\section{Заключение}
\begin{itemize}
    \item Магнитный момент неодимового шарика: $\mm = \res{muSI}$~А$\cdot$м$^2$;

\end{itemize}
